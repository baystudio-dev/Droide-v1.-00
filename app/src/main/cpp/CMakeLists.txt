cmake_minimum_required(VERSION 3.22.1)
project(droide_terminal C)
add_library(termux SHARED ../../../../third_party/termux/native/termux.c)
target_compile_options(termux PRIVATE -Os -Wall -Wextra)
target_link_options(termux PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
add_library(droide_files SHARED droide_files.c)
target_compile_options(droide_files PRIVATE -Os -Wall -Wextra -Werror)
target_link_options(droide_files PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
